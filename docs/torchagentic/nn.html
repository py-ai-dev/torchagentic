544    def read(
545        self,
546        query: torch.Tensor,
547        state: Optional[MemoryState] = None,
548    ) -> Tuple[torch.Tensor, MemoryState]:
549        """Read from memory without writing.
550
551        Args:
552            query: (batch, slot_size) or (batch, seq_len, slot_size)
553            state: Current memory state (uses internal state if None).
554
555        Returns:
556            (read_vectors, new_state)
557        """
558        B, L = query.shape[0], 1 if query.dim() == 2 else query.shape[1]
559        if query.dim() == 2:
560            query = query.unsqueeze(1)
561
562        state = state if state is not None else self._state
563        if state is None:
564            state = self.reset(B)
565
566        # Backend produces read weights via content/location addressing.
567        read_w, _write_params, new_s = self.backend(query[:, -1, :], state)
568        vectors = torch.matmul(read_w, new_s.memory.unsqueeze(0))

570        self._state = new_s

573    def write(
574        self,
575        key: torch.Tensor,
576        value: torch.Tensor,
577        state: Optional[MemoryState] = None,
578    ) -> MemoryState:
579        """Write to memory.
580
581        Args:
582            key:   (batch, slot_size) — determines where to write.
583            value: (batch, slot_size) — content to write.
584            state: Current memory state.
585
586        Returns:
587            new_state  (memory matrix is updated)
588        """
589        B = key.shape[0]
590        state = state if state is not None else self._state
591        if state is None:
592            state = self.reset(B)
593
594        # Use a dummy read to get write params from the backend.
595        _read_w, write_params, new_s = self.backend(key, state)
596
597        # Parse write params.
598        W = self.num_writes
599        ws = self.slot_size
600        write_vals = write_params[:, :W * ws].reshape(B, W, ws)
601        erase = torch.sigmoid(write_params[:, W * ws:].reshape(B, W, ws))
602
603        # Erase.
604        erase_mat = (new_s.write_weights.transpose(1, 2) @ erase).mean(dim=0)
605        memory = new_s.memory * (1 - erase_mat)
606
607        # Add.
608        add_mat = (new_s.write_weights.transpose(1, 2) @ value.unsqueeze(1)).mean(dim=0)
609        memory = memory + add_mat
610
611        new_s.memory = memory
612        self._state = new_s
613        return new_s

615    def forward(
616        self,
617        x: torch.Tensor,
618        state: Optional[MemoryState] = None,
619    ) -> Tuple[torch.Tensor, MemoryState]:
620        """One step: attend to memory → write → read.
621
622        Equivalent to: read(write(query)) as in the original NTM.
623
624        Args:
625            x:     (batch, slot_size) or (batch, seq_len, slot_size)
626            state: Optional initial state.
627
628        Returns:
629            (read_vectors, new_state)
630        """
631        B, L = x.shape[0], 1 if x.dim() == 2 else x.shape[1]
632        flat = x.dim() == 2
633
634        state = state if state is not None else self._state
635        if state is None:
636            state = self.reset(B)
637
638        outputs = []
639        for t in range(L):
640            step = x[:, t, :] if not flat else x
641            rw, wp, state = self.backend(step, state)
642            # Write: erase then add.
643            W, S = self.num_writes, self.slot_size
644            write_vals = wp[:, :W * S].reshape(B, W, S)
645            erase = torch.sigmoid(wp[:, W * S:].reshape(B, W, S))
646            erase_mat = (state.write_weights.transpose(1, 2) @ erase).mean(dim=0)
647            state.memory = state.memory * (1 - erase_mat)
648            add_mat = (state.write_weights.transpose(1, 2) @ write_vals).mean(dim=0)
649            state.memory = state.memory + add_mat
650            # Read.
651            vectors = torch.matmul(rw, state.memory.unsqueeze(0))
652            outputs.append(vectors)
653
654        result = torch.stack(outputs, dim=1) if L > 1 else outputs[0]
655        self._state = state
656        return result, state

152        mem_norm = F.normalize(memory.unsqueeze(0), dim=-1)
153        key_norm = F.normalize(key.unsqueeze(1), dim=-1)
154        sim = torch.matmul(key_norm, mem_norm.transpose(-2, -1))

158    def _convolve(weights: torch.Tensor, shift: torch.Tensor) -> torch.Tensor:
159        padded = F.pad(weights, (1, 1), mode="circular")
160        return (
161            padded[..., :-2] * shift[..., 0:1]
162            + padded[..., 1:-1] * shift[..., 1:2]
163            + padded[..., 2:] * shift[..., 2:3]
164        )

167        self,
168        query: torch.Tensor,
169        state: MemoryState,

174        off = 0

176        off += self.slot_size

178        off += 1

281        usage = usage.unsqueeze(0)
282        idx = torch.argsort(usage, dim=-1, descending=False)
283        sorted_u = torch.gather(usage, -1, idx)
284        cumprod = torch.cumprod(1 - sorted_u, dim=-1)
285        alloc = torch.zeros_like(sorted_u)
286        alloc[..., 0] = sorted_u[..., 0]
287        alloc[..., 1:] = sorted_u[..., 1:] * cumprod[..., :-1]
288        # Unsort.
289        inv_idx = torch.argsort(idx, dim=-1)
290        alloc = torch.gather(alloc, -1, inv_idx)
291        return alloc * gate.unsqueeze(-1)
