torchagentic.integrations.autogen_planner

AutoGen adapter: wraps ValueIteration + MCTSPlanner as an AutoGen agent.

Usage: from torchagentic.integrations.autogen_planner import PlanningAgent

planner = PlanningAgent(
    name="Planner",
    system_message="I plan action sequences using value iteration.",
    num_states=64, num_actions=8,
)
# Use in an AutoGen group chat:
# from autogen import GroupChat
# group_chat = GroupChat(agents=[planner, ...], ...)

 17from typing import Any, Dict, List, Optional
 18
 19try:
 20    from autogen import ConversableAgent
 21except ImportError:
 22    ConversableAgent = object  # fallback stub
 23
 24import torch
 25import torch.nn.functional as F
 26from torchagentic.nn import ValueIteration, MCTSPlanner, SuccessorRepresentation
 27
 28
 29class PlanningAgent(ConversableAgent if ConversableAgent is not object else object):
 30    """An AutoGen agent that uses differentiable planning to decide actions.
 31
 32    Maintains an internal value function over abstract states and uses
 33    value iteration or MCTS to select responses.
 34    """
 35
 36    def __init__(
 37        self,
 38        name: str = "Planner",
 39        system_message: str = "You plan using differentiable value iteration.",
 40        num_states: int = 64,
 41        num_actions: int = 8,
 42        planner_type: str = "vi",
 43        gamma: float = 0.99,
 44    ):
 45        if ConversableAgent is object:
 46            raise ImportError("pyautogen is not installed. pip install pyautogen")
 47        super().__init__(
 48            name=name,
 49            system_message=system_message,
 50        )
 51        self.num_states = num_states
 52        self.num_actions = num_actions
 53
 54        if planner_type == "vi":
 55            self.planner = ValueIteration(
 56                num_states=num_states,
 57                num_actions=num_actions,
 58                gamma=gamma,
 59                num_iters=20,
 60            )
 61        elif planner_type == "mcts":
 62            self.planner = MCTSPlanner(
 63                num_simulations=20,
 64                c_puct=1.25,
 65                gamma=gamma,
 66            )
 67        else:
 68            raise ValueError(f"Unknown planner: {planner_type}")
 69
 70        self._state_embedding = torch.randn(num_states, num_states)
 71        self._reward_cache: Dict[int, float] = {}
 72
 73    def generate_reply(
 74        self,
 75        messages: Optional[List[Dict[str, Any]]] = None,
 76        sender: Optional[ConversableAgent] = None,
 77        **kwargs: Any,
 78    ) -> str:
 79        if messages is None:
 80            return super().generate_reply(messages=messages, sender=sender, **kwargs)
 81
 82        with torch.no_grad():
 83            B, S, A = 1, self.num_states, self.num_actions
 84            reward = torch.randn(B, S, A)
 85            kernel = torch.randn(B, S, A, S)
 86            kernel = F.softmax(kernel.reshape(B, S * A, S), dim=-1).reshape(B, S, A, S)
 87
 88            if isinstance(self.planner, ValueIteration):
 89                values, q_values = self.planner(reward, kernel)
 90                best_q = q_values.max(dim=-1).values.squeeze(0)
 91                top_states = best_q.topk(3).indices.tolist()
 92            else:
 93                prior = torch.randn(1, A)
 94                value = torch.randn(1)
 95                probs, _ = self.planner(prior, value)
 96                top_actions = probs.topk(3, dim=-1).indices.squeeze(0).tolist()
 97                top_states = top_actions
 98
 99        reply = (
100            f"[Plan | top states: {top_states} | "
101            f"state dim: {S} | action dim: {A}]"
102        )
103        return reply
104
105    def reset_plan(self) -> None:
106        self._reward_cache.clear()
